import { Card, CardGrid } from '@astrojs/starlight/components';

<CardGrid stagger>
  <Card title="Compiled, not interpreted" icon="rocket">
    The compiler does the heavy lifting up front: layout, reactivity and render calls are
    worked out at build time and lowered through LLVM to native code.
  </Card>
  <Card title="Predictable state" icon="random">
    Every UI follows the Model-View-Update contract: `init`, `update`, `view`. State is
    plain data and `view` is always pure.
  </Card>
  <Card title="Embeds in your engine" icon="puzzle">
    Kura compiles to an object or library you link into your existing project. No manifest,
    no framework, no browser.
  </Card>
  <Card title="Seamless C interop" icon="seti:c">
    Call C headers directly and export Kura functions as C functions. No bindings to write.
  </Card>
</CardGrid>
