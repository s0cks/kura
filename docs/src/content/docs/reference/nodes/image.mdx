A <Node name="box" /> for rendering an image or texture.

<Example>

```kura
#image(
  width: 120px,
  height: 120px,
  fit: "contain",
  "assets/avatar.png",
)
```

</Example>

<PropertyTable inherits="box">
  <Property name="fit" types={["String"]} initial='"fill"'>
    How the image scales inside its box: `"contain"`, `"cover"`, `"fill"`, or `"none"`.
  </Property>
  <TableSeparator text="Positional Properties" color="primary" />
  <Property name="source" types={["String"]}>
    The source URI or path for the image.
  </Property>
</PropertyTable>
